# The optional Wayland backend uses public Qt 6.10 native interfaces and locally installed
# protocol definitions. All other platforms retain the portable capture implementation.
option(LUX_UI_ENABLE_WAYLAND_CAPTURE "Enable public Wayland relative pointer capture when available" ON)
if(LUX_UI_ENABLE_WAYLAND_CAPTURE AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND Qt6Gui_VERSION VERSION_GREATER_EQUAL 6.10)
    find_package(Qt6WaylandClient QUIET CONFIG)
    find_package(PkgConfig QUIET)
    find_program(LUXUI_WAYLAND_SCANNER NAMES wayland-scanner)
    if(PkgConfig_FOUND)
        pkg_check_modules(LUXUI_WAYLAND_CLIENT QUIET IMPORTED_TARGET wayland-client)
        pkg_get_variable(LUXUI_WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
    endif()
    set(_luxui_relative_xml "${LUXUI_WAYLAND_PROTOCOLS_DIR}/unstable/relative-pointer/relative-pointer-unstable-v1.xml")
    set(_luxui_constraints_xml "${LUXUI_WAYLAND_PROTOCOLS_DIR}/unstable/pointer-constraints/pointer-constraints-unstable-v1.xml")
    if(TARGET Qt6::WaylandClient AND TARGET PkgConfig::LUXUI_WAYLAND_CLIENT
       AND LUXUI_WAYLAND_SCANNER AND EXISTS "${_luxui_relative_xml}" AND EXISTS "${_luxui_constraints_xml}")
        include(CheckCXXSourceCompiles)
        include(CMakePushCheckState)
        cmake_push_check_state(RESET)
        set(CMAKE_REQUIRED_LIBRARIES Qt6::Gui)
        check_cxx_source_compiles("#include <QGuiApplication>
#include <QtGui/qguiapplication_platform.h>
int main() { QGuiApplication* app = nullptr; return app ? (app->nativeInterface<QNativeInterface::QWaylandApplication>() != nullptr) : 0; }"
            LUXUI_HAS_PUBLIC_WAYLAND_INTERFACE)
        cmake_pop_check_state()
        if(LUXUI_HAS_PUBLIC_WAYLAND_INTERFACE)
            set(_luxui_protocol_dir "${CMAKE_CURRENT_BINARY_DIR}/wayland-protocols")
            file(MAKE_DIRECTORY "${_luxui_protocol_dir}")
            set(_luxui_protocol_files)
            foreach(_protocol relative-pointer pointer-constraints)
                if(_protocol STREQUAL "relative-pointer")
                    set(_xml "${_luxui_relative_xml}")
                else()
                    set(_xml "${_luxui_constraints_xml}")
                endif()
                set(_header "${_luxui_protocol_dir}/${_protocol}-client-protocol.h")
                set(_code "${_luxui_protocol_dir}/${_protocol}-protocol.inc")
                add_custom_command(OUTPUT "${_header}" "${_code}"
                    COMMAND "${LUXUI_WAYLAND_SCANNER}" client-header "${_xml}" "${_header}"
                    COMMAND "${LUXUI_WAYLAND_SCANNER}" private-code "${_xml}" "${_code}"
                    DEPENDS "${_xml}" VERBATIM)
                list(APPEND _luxui_protocol_files "${_header}" "${_code}")
            endforeach()
            add_custom_target(LuxUIWaylandProtocolFiles DEPENDS ${_luxui_protocol_files})
            add_dependencies(LuxUI LuxUIWaylandProtocolFiles)
            set_source_files_properties(${_luxui_protocol_files} PROPERTIES SKIP_AUTOMOC ON)
            target_sources(LuxUI PRIVATE Private/PointerCaptureWayland.cpp
                Private/WaylandProtocols.cpp ${_luxui_protocol_files})
            target_include_directories(LuxUI PRIVATE "${_luxui_protocol_dir}")
            target_compile_definitions(LuxUI PRIVATE LUX_UI_WAYLAND_CAPTURE=1)
            target_link_libraries(LuxUI PRIVATE Qt6::WaylandClient PkgConfig::LUXUI_WAYLAND_CLIENT)
            message(STATUS "LuxUI Wayland relative pointer capture: enabled")
        endif()
    endif()
endif()
