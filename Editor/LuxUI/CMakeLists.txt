set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
add_library(LuxUI STATIC Private/Framework.cpp Private/Panels.cpp Private/AssetList.cpp Private/Window.cpp Public/LuxUI/Widgets.hpp)
add_library(LuxUI::LuxUI ALIAS LuxUI)
set_target_properties(LuxUI PROPERTIES AUTOMOC ON)
target_include_directories(LuxUI PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Public")
target_compile_features(LuxUI PUBLIC cxx_std_20)
target_link_libraries(LuxUI PUBLIC Qt6::Widgets)
if(MSVC)
    target_compile_options(LuxUI PUBLIC /utf-8)
endif()

if(LUX_BUILD_TESTS)
    add_executable(LuxUIFrameworkTests Tests/FrameworkTest.cpp)
    target_link_libraries(LuxUIFrameworkTests PRIVATE LuxUI)
    add_executable(LuxUIPanelsTests Tests/PanelsTest.cpp)
    target_link_libraries(LuxUIPanelsTests PRIVATE LuxUI)
    add_test(NAME LuxUI_panels COMMAND LuxUIPanelsTests)
    set_tests_properties(LuxUI_panels PROPERTIES LABELS "headless;editor;luxui" TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME LuxUI_framework COMMAND LuxUIFrameworkTests)
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_test(NAME LuxUI_boundaries COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/VerifyEditorBoundaries.py")
    set_tests_properties(LuxUI_boundaries PROPERTIES LABELS "headless;editor;luxui" TIMEOUT 30)
    set_tests_properties(LuxUI_framework PROPERTIES LABELS "headless;editor;luxui" TIMEOUT 30)
endif()
