if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.24)
    project(ForestSurvivor LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    option(LUX_BUILD_TESTS "Build sample gameplay tests" ON)
    if(LUX_BUILD_TESTS)
        enable_testing()
    endif()
    find_package(LuxEngine CONFIG REQUIRED)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

add_library(SampleForestSurvivorGame STATIC Game/SurvivalGame.cpp)
target_include_directories(SampleForestSurvivorGame PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(SampleForestSurvivorGame PUBLIC LuxEngine::Engine)
if(COMMAND lux_set_target_warnings)
    lux_set_target_warnings(SampleForestSurvivorGame)
elseif(MSVC)
    target_compile_options(SampleForestSurvivorGame PRIVATE /utf-8)
endif()
if(LUX_BUILD_TESTS)
    add_executable(SampleForestSurvivorTests Tests/SurvivalGameTest.cpp)
    target_link_libraries(SampleForestSurvivorTests PRIVATE SampleForestSurvivorGame)
    if(COMMAND lux_set_target_warnings)
        lux_set_target_warnings(SampleForestSurvivorTests)
    endif()
    add_test(NAME Sample_forest_survivor_rules COMMAND SampleForestSurvivorTests)
    set_tests_properties(Sample_forest_survivor_rules PROPERTIES LABELS "headless;samples" TIMEOUT 60)

endif()
